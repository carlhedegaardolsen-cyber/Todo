import tkinter as tk from tkinter import ttk import json import os import sys import winreg from datetime import datetime # ========================================================== # SETTINGS # ========================================================== APP_NAME = "My Todo" DATA_FILE = os.path.join( os.path.dirname(os.path.abspath(__file__)), "todos.json" ) # ========================================================== # DATA # ========================================================== todos = [] fitness_logs = [] def load_data(): global todos, fitness_logs if not os.path.exists(DATA_FILE): return try: with open(DATA_FILE, "r", encoding="utf-8") as file: data = json.load(file) # Support older versions if isinstance(data, list): todos = data fitness_logs = [] else: todos = data.get("todos", []) fitness_logs = data.get("fitness_logs", []) except (json.JSONDecodeError, OSError): todos = [] fitness_logs = [] def save_data(): data = { "todos": todos, "fitness_logs": fitness_logs } try: with open(DATA_FILE, "w", encoding="utf-8") as file: json.dump( data, file, indent=4, ensure_ascii=False ) except OSError: pass # ========================================================== # DATE # ========================================================== def get_date(): return datetime.now().strftime( "%A, %d %B %Y" ) def get_short_date(): return datetime.now().strftime( "%d/%m/%Y" ) # ========================================================== # TODO FUNCTIONS # ========================================================== def add_todo(event=None): text = todo_entry.get().strip() if not text: return todos.append({ "text": text, "completed": False }) todo_entry.delete( 0, tk.END ) save_data() refresh_todos() todo_entry.focus() def complete_selected(event=None): selection = todo_list.curselection() if not selection: return index = selection[0] todos[index]["completed"] = True save_data() refresh_todos() def uncomplete_selected(event=None): selection = completed_list.curselection() if not selection: return index = selection[0] completed_items = [ todo for todo in todos if todo.get("completed", False) ] selected_todo = completed_items[index] selected_todo["completed"] = False save_data() refresh_todos() def delete_selected(event=None): # If an active todo is selected selection = todo_list.curselection() if selection: index = selection[0] todos.pop(index) save_data() refresh_todos() return # If a completed todo is selected selection = completed_list.curselection() if selection: index = selection[0] completed_items = [ todo for todo in todos if todo.get("completed", False) ] selected_todo = completed_items[index] todos.remove(selected_todo) save_data() refresh_todos() def clear_completed(): global todos todos = [ todo for todo in todos if not todo.get("completed", False) ] save_data() refresh_todos() # ========================================================== # EDIT TODO # ========================================================== def edit_selected(event=None): selection = todo_list.curselection() if not selection: return index = selection[0] open_edit_window(index) def open_edit_window(index): edit_window = tk.Toplevel(window) edit_window.title( "Edit Todo" ) edit_window.geometry( "430x170" ) edit_window.configure( bg="white" ) edit_window.resizable( False, False ) edit_window.transient( window ) edit_window.grab_set() label = tk.Label( edit_window, text="Edit task", font=("Segoe UI", 11), bg="white", fg="#333333" ) label.pack( pady=(22, 7) ) edit_entry = tk.Entry( edit_window, font=("Segoe UI", 13), relief="solid", bd=1 ) edit_entry.pack( fill="x", padx=30, ipady=5 ) edit_entry.insert( 0, todos[index]["text"] ) edit_entry.select_range( 0, tk.END ) edit_entry.focus() def save_edit(event=None): new_text = edit_entry.get().strip() if new_text: todos[index]["text"] = new_text save_data() refresh_todos() edit_window.destroy() edit_entry.bind( "", save_edit ) save_button = tk.Button( edit_window, text="Save", command=save_edit, font=("Segoe UI", 10, "bold"), bg="#222222", fg="white", activebackground="#444444", activeforeground="white", relief="flat", padx=25, pady=7 ) save_button.pack( pady=15 ) # ========================================================== # REFRESH TODO LISTS # ========================================================== def refresh_todos(): todo_list.delete( 0, tk.END ) completed_list.delete( 0, tk.END ) active_count = 0 completed_count = 0 # Active tasks for todo in todos: if not todo.get("completed", False): todo_list.insert( tk.END, "○ " + todo.get("text", "") ) active_count += 1 # Completed tasks for todo in todos: if todo.get("completed", False): completed_list.insert( tk.END, "✓ " + todo.get("text", "") ) completed_count += 1 # Counter if active_count == 1: task_count_label.config( text="1 task remaining" ) else: task_count_label.config( text=f"{active_count} tasks remaining" ) if completed_count == 1: completed_count_label.config( text="1 completed" ) else: completed_count_label.config( text=f"{completed_count} completed" ) # ========================================================== # SEARCH # ========================================================== def search_todos(event=None): search_text = ( search_entry .get() .lower() .strip() ) todo_list.delete( 0, tk.END ) completed_list.delete( 0, tk.END ) for todo in todos: text = todo.get( "text", "" ) if search_text in text.lower(): if todo.get("completed", False): completed_list.insert( tk.END, "✓ " + text ) else: todo_list.insert( tk.END, "○ " + text ) def clear_search(): search_entry.delete( 0, tk.END ) refresh_todos() search_entry.focus() # ========================================================== # FITNESS # ========================================================== def add_fitness(event=None): text = fitness_entry.get().strip() if not text: return fitness_logs.append({ "date": get_short_date(), "text": text }) fitness_entry.delete( 0, tk.END ) save_data() refresh_fitness() fitness_entry.focus() def delete_fitness(event=None): selection = fitness_list.curselection() if not selection: return displayed_index = selection[0] actual_index = ( len(fitness_logs) - 1 - displayed_index ) fitness_logs.pop( actual_index ) save_data() refresh_fitness() def refresh_fitness(): fitness_list.delete( 0, tk.END ) for log in reversed(fitness_logs): display_text = ( log.get("date", "") + " " + log.get("text", "") ) fitness_list.insert( tk.END, display_text ) total = len(fitness_logs) if total == 1: fitness_count_label.config( text="1 workout entry" ) else: fitness_count_label.config( text=f"{total} workout entries" ) # ========================================================== # WINDOWS STARTUP # ========================================================== def enable_startup(): try: script_path = os.path.abspath( sys.argv[0] ) python_exe = sys.executable.replace( "python.exe", "pythonw.exe" ) command = ( f'"{python_exe}" ' f'"{script_path}"' ) key = winreg.OpenKey( winreg.HKEY_CURRENT_USER, r"Software\Microsoft\Windows\CurrentVersion\Run", 0, winreg.KEY_SET_VALUE ) winreg.SetValueEx( key, APP_NAME, 0, winreg.REG_SZ, command ) winreg.CloseKey( key ) except Exception: pass # ========================================================== # MAIN WINDOW # ========================================================== load_data() window = tk.Tk() window.title( "My Todo" ) window.geometry( "680x760" ) window.minsize( 580, 650 ) window.configure( bg="white" ) # ========================================================== # STYLE # ========================================================== style = ttk.Style() try: style.theme_use( "clam" ) except: pass style.configure( "TNotebook", background="white", borderwidth=0 ) style.configure( "TNotebook.Tab", background="white", foreground="#888888", padding=(20, 10), font=("Segoe UI", 10) ) style.map( "TNotebook.Tab", foreground=[ ("selected", "#222222") ] ) # ========================================================== # MAIN CONTAINER # ========================================================== main = tk.Frame( window, bg="white" ) main.pack( fill="both", expand=True, padx=45, pady=35 ) # ========================================================== # DATE # ========================================================== date_label = tk.Label( main, text=get_date(), font=("Segoe UI", 10), bg="white", fg="#999999" ) date_label.pack( anchor="w", pady=(0, 12) ) # ========================================================== # TABS # ========================================================== notebook = ttk.Notebook( main ) notebook.pack( fill="both", expand=True ) # ========================================================== # TODOS TAB # ========================================================== todo_tab = tk.Frame( notebook, bg="white" ) notebook.add( todo_tab, text="Todos" ) todo_container = tk.Frame( todo_tab, bg="white" ) todo_container.pack( fill="both", expand=True, padx=5, pady=10 ) # ========================================================== # TODO TITLE # ========================================================== todo_title = tk.Label( todo_container, text="My Todos", font=("Segoe UI", 28, "bold"), bg="white", fg="#171717" ) todo_title.pack( anchor="w", pady=(0, 5) ) todo_subtitle = tk.Label( todo_container, text="What needs to get done?", font=("Segoe UI", 11), bg="white", fg="#888888" ) todo_subtitle.pack( anchor="w", pady=(0, 22) ) # ========================================================== # ADD TODO # ========================================================== todo_input_frame = tk.Frame( todo_container, bg="white" ) todo_input_frame.pack( fill="x", pady=(0, 12) ) todo_entry = tk.Entry( todo_input_frame, font=("Segoe UI", 14), relief="solid", bd=1 ) todo_entry.pack( side="left", fill="x", expand=True, ipady=10 ) add_todo_button = tk.Button( todo_input_frame, text="Add", command=add_todo, font=("Segoe UI", 11, "bold"), bg="#222222", fg="white", activebackground="#444444", activeforeground="white", relief="flat", padx=22, pady=9 ) add_todo_button.pack( side="left", padx=(10, 0) ) # ========================================================== # SEARCH # ========================================================== search_frame = tk.Frame( todo_container, bg="white" ) search_frame.pack( fill="x", pady=(0, 12) ) search_entry = tk.Entry( search_frame, font=("Segoe UI", 10), relief="solid", bd=1 ) search_entry.pack( side="left", fill="x", expand=True, ipady=6 ) search_button = tk.Button( search_frame, text="Search", command=search_todos, font=("Segoe UI", 9), bg="white", fg="#555555", activebackground="#eeeeee", relief="flat", bd=0, padx=10 ) search_button.pack( side="left" ) clear_search_button = tk.Button( search_frame, text="Clear", command=clear_search, font=("Segoe UI", 9), bg="white", fg="#999999", activebackground="#eeeeee", relief="flat", bd=0 ) clear_search_button.pack( side="left" ) # ========================================================== # ACTIVE SECTION # ========================================================== active_label = tk.Label( todo_container, text="Tasks", font=("Segoe UI", 13, "bold"), bg="white", fg="#333333" ) active_label.pack( anchor="w", pady=(5, 5) ) todo_list = tk.Listbox( todo_container, font=("Segoe UI", 13), bg="white", fg="#222222", selectbackground="#eeeeee", selectforeground="#222222", highlightthickness=0, relief="flat", bd=0, activestyle="none" ) todo_list.pack( fill="both", expand=True ) # ========================================================== # COMPLETED SECTION # ========================================================== completed_header = tk.Frame( todo_container, bg="white" ) completed_header.pack( fill="x", pady=(20, 5) ) completed_label = tk.Label( completed_header, text="Completed", font=("Segoe UI", 13, "bold"), bg="white", fg="#777777" ) completed_label.pack( side="left" ) completed_count_label = tk.Label( completed_header, text="0 completed", font=("Segoe UI", 9), bg="white", fg="#aaaaaa" ) completed_count_label.pack( side="right" ) completed_list = tk.Listbox( todo_container, font=("Segoe UI", 12), bg="white", fg="#999999", selectbackground="#eeeeee", selectforeground="#777777", highlightthickness=0, relief="flat", bd=0, activestyle="none", height=5 ) completed_list.pack( fill="x" ) # ========================================================== # TODO FOOTER # ========================================================== todo_footer = tk.Frame( todo_container, bg="white" ) todo_footer.pack( fill="x", pady=(12, 0) ) task_count_label = tk.Label( todo_footer, text="0 tasks remaining", font=("Segoe UI", 9), bg="white", fg="#999999" ) task_count_label.pack( side="left" ) clear_completed_button = tk.Button( todo_footer, text="Clear completed", command=clear_completed, font=("Segoe UI", 9), bg="white", fg="#555555", activebackground="#eeeeee", relief="flat", bd=0 ) clear_completed_button.pack( side="right" ) delete_button = tk.Button( todo_footer, text="Delete selected", command=delete_selected, font=("Segoe UI", 9), bg="white", fg="#555555", activebackground="#eeeeee", relief="flat", bd=0 ) delete_button.pack( side="right", padx=(0, 15) ) # ========================================================== # FITNESS TAB # ========================================================== fitness_tab = tk.Frame( notebook, bg="white" ) notebook.add( fitness_tab, text="Fitness" ) fitness_container = tk.Frame( fitness_tab, bg="white" ) fitness_container.pack( fill="both", expand=True, padx=5, pady=10 ) fitness_title = tk.Label( fitness_container, text="Fitness", font=("Segoe UI", 28, "bold"), bg="white", fg="#171717" ) fitness_title.pack( anchor="w", pady=(0, 5) ) fitness_subtitle = tk.Label( fitness_container, text="Keep track of what you achieved.", font=("Segoe UI", 11), bg="white", fg="#888888" ) fitness_subtitle.pack( anchor="w", pady=(0, 22) ) # ========================================================== # FITNESS INPUT # ========================================================== fitness_input_frame = tk.Frame( fitness_container, bg="white" ) fitness_input_frame.pack( fill="x", pady=(0, 20) ) fitness_entry = tk.Entry( fitness_input_frame, font=("Segoe UI", 13), relief="solid", bd=1 ) fitness_entry.pack( side="left", fill="x", expand=True, ipady=10 ) fitness_button = tk.Button( fitness_input_frame, text="Log", command=add_fitness, font=("Segoe UI", 10, "bold"), bg="#222222", fg="white", activebackground="#444444", activeforeground="white", relief="flat", padx=22, pady=9 ) fitness_button.pack( side="left", padx=(10, 0) ) fitness_hint = tk.Label( fitness_container, text="Examples: Bench 60kg × 8 • Run 5km • Squat 80kg × 5", font=("Segoe UI", 9), bg="white", fg="#aaaaaa" ) fitness_hint.pack( anchor="w", pady=(0, 15) ) # ========================================================== # FITNESS LIST # ========================================================== fitness_list = tk.Listbox( fitness_container, font=("Segoe UI", 12), bg="white", fg="#444444", selectbackground="#eeeeee", selectforeground="#222222", highlightthickness=0, relief="flat", bd=0, activestyle="none", height=15 ) fitness_list.pack( fill="both", expand=True ) # ========================================================== # FITNESS FOOTER # ========================================================== fitness_footer = tk.Frame( fitness_container, bg="white" ) fitness_footer.pack( fill="x", pady=(12, 0) ) fitness_count_label = tk.Label( fitness_footer, text="0 workout entries", font=("Segoe UI", 9), bg="white", fg="#999999" ) fitness_count_label.pack( side="left" ) delete_fitness_button = tk.Button( fitness_footer, text="Delete selected", command=delete_fitness, font=("Segoe UI", 9), bg="white", fg="#555555", activebackground="#eeeeee", relief="flat", bd=0 ) delete_fitness_button.pack( side="right" ) #
